let model = null;
let aiRunning = false;
let lastDetection = 0;

async function loadAIModel() {
    try {
        document.getElementById("aiStatus").innerText =
            "🤖 Loading YOLO model...";

        model = await tf.loadGraphModel("/model/model.json");

        document.getElementById("aiStatus").innerText =
            "✅ YOLO model loaded";
    } catch (error) {
        console.error(error);

        document.getElementById("aiStatus").innerText =
            "❌ Could not load YOLO model";
    }
}

async function startAIDetection() {
    if (!model) {
        alert("YOLO model is not loaded.");
        return;
    }

    if (!cameraStream) {
        alert("Start camera first.");
        return;
    }

    if (latitude === null) {
        alert("Start GPS first.");
        return;
    }

    aiRunning = true;

    document.getElementById("aiStatus").innerText =
        "🔍 AI is detecting potholes...";

    detectFrame();
}

async function detectFrame() {

    if (!aiRunning) return;

    const video = document.getElementById("camera");

    if (video.readyState < 2) {
        requestAnimationFrame(detectFrame);
        return;
    }

    const input = tf.browser
        .fromPixels(video)
        .resizeBilinear([640, 640])
        .toFloat()
        .div(255.0)
        .expandDims(0);

    try {

        const output = await model.executeAsync(input);

        /*
          IMPORTANT:
          The exact output processing depends on
          how your YOLO model was exported.
        */

        const potholeFound = processYOLOOutput(output);

        if (potholeFound) {

            const now = Date.now();

            // Prevent sending the same pothole repeatedly
            if (now - lastDetection > 5000) {

                lastDetection = now;

                sendPotholeAlert();
            }
        }

        if (Array.isArray(output)) {
            output.forEach(t => t.dispose());
        } else {
            output.dispose();
        }

    } catch (error) {
        console.error("YOLO error:", error);
    }

    input.dispose();

    // Run detection again
    setTimeout(detectFrame, 200);
}
